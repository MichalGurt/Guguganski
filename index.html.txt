<textarea id="output" placeholder="Wynik..."></textarea><script>
const alfabetBig = {
  'A':'GU','Ą':'GS','B':'GA','C':'GE','Ć':'GY','D':'GI','E':'GW','Ę':'GF','F':'GH','G':'GG','H':'GL',
  'I':'GR','J':'GT','K':'GP','L':'GZ','Ł':'GX','M':'GV','N':'GN','Ń':'GM','O':'GO','Ó':'GQ',
  'P':'GK','R':'GD','S':'GB','Ś':'GJ','T':'GUU','U':'GAA','W':'GEE','Y':'GII','Z':'GOO','Ź':'GYY','Ż':'GZZ'
};

const alfabetSmall = {};
for (let key in alfabetBig) {
  alfabetSmall[key.toLowerCase()] = alfabetBig[key].toLowerCase();
}

const reverseBig = {};
const reverseSmall = {};

for (let key in alfabetBig) {
  reverseBig[alfabetBig[key]] = key;
}

for (let key in alfabetSmall) {
  reverseSmall[alfabetSmall[key]] = key;
}

function toGuguganski() {
  let text = document.getElementById('input').value;
  let result = '';

  for (let char of text) {
    if (alfabetBig[char]) {
      result += alfabetBig[char];
    } else if (alfabetSmall[char]) {
      result += alfabetSmall[char];
    } else if (char === ' ') {
      result += ' ';
    } else {
      result += char;
    }
  }

  document.getElementById('output').value = result;
}

function toPolski() {
  let text = document.getElementById('input').value;
  let result = '';
  let i = 0;

  while (i < text.length) {
    let found = false;

    // sprawdzanie dużych
    for (let key in reverseBig) {
      if (text.startsWith(key, i)) {
        result += reverseBig[key];
        i += key.length;
        found = true;
        break;
      }
    }

    if (found) continue;

    // sprawdzanie małych
    for (let key in reverseSmall) {
      if (text.startsWith(key, i)) {
        result += reverseSmall[key];
        i += key.length;
        found = true;
        break;
      }
    }

    if (!found) {
      result += text[i];
      i++;
    }
  }

  document.getElementById('output').value = result;
}
